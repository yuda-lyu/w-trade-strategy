import fs from 'fs'
import path from 'path'
import get from 'lodash-es/get.js'
import isEqual from 'lodash-es/isEqual.js'
import map from 'lodash-es/map.js'
import isobj from 'wsemi/src/isobj.mjs'
import isfun from 'wsemi/src/isfun.mjs'
import isenum from 'wsemi/src/isenum.mjs'
import cdbl from 'wsemi/src/cdbl.mjs'
import istime from 'wsemi/src/istime.mjs'
import dataProvide from 'w-data-tdprovide/src/WDataTdprovide.mjs'
import runStrategy from 'w-trade-backtest/src/runStrategy.mjs'
import calcFitness from 'w-trade-solve/src/calcFitness.mjs'
import genStrategyFileName from 'w-trade-solve/src/genStrategyFileName.mjs'
import p2r from 'w-trade-solve/src/p2r.mjs'
import { toNumericUnit } from '../cfg.mjs'
import { makeOtt } from '../util/ott.mjs'


/**
 * 策略庫重算時逐隻斷言不變之摘要欄(交易筆數、已平倉筆數、勝率)
 *
 * @type {Array}
 */
let INVARIANT_KEYS = Object.freeze(['numTrade', 'numTradeFin', 'rWin'])


/**
 * 比對兩份回測摘要之逐欄差異(timeTest為回測執行時刻, 不比)
 *
 * 各欄以JSON序列化後比對(數值以最短十進位表示, 與策略檔之內容同口徑)；一方缺欄亦列為差異
 *
 * @function
 * @param {Object} a 輸入摘要物件(例如策略檔所存者)
 * @param {Object} b 輸入摘要物件(例如重放所得者)
 * @returns {Array} 回傳不同之欄名字串陣列(依a之欄序, 其後為b獨有之欄)
 * @example
 *
 * console.log(diffSummary({ timeTest: 't1', numTrade: 3, rWin: '50.00%' }, { timeTest: 't2', numTrade: 3, rWin: '66.67%', uTradeAllMax: 2 }))
 * // => [ 'rWin', 'uTradeAllMax' ]
 *
 */
let diffSummary = (a, b) => {
    let ks = [...Object.keys(a || {}), ...Object.keys(b || {}).filter((k) => !Object.prototype.hasOwnProperty.call(a || {}, k))]
    return ks.filter((k) => k !== 'timeTest' && JSON.stringify(get(a, k)) !== JSON.stringify(get(b, k)))
}


/**
 * 取組策略檔名所用之交易次數級距設定(檔內solve.levelNumTradeBounds；以w-trade-solve之opt.levelNumTradeBounds求得之策略記之，未記者為null即預設級距)
 *
 * @function
 * @param {Object} data 輸入策略檔內容
 * @returns {Object} 回傳genStrategyFileName之opt物件{levelNumTradeBounds}
 * @example
 *
 * console.log(optFnOf({ solve: { levelNumTradeBounds: [1, 3, 6] } }), optFnOf({ solve: {} }))
 * // => { levelNumTradeBounds: [ 1, 3, 6 ] } { levelNumTradeBounds: null }
 *
 */
let optFnOf = (data) => ({ levelNumTradeBounds: get(data, 'solve.levelNumTradeBounds', null) })


/**
 * 重算前後之識別不變檢查: 交易筆數、已平倉筆數、勝率(INVARIANT_KEYS)須與原摘要相同，且以新摘要所組之檔名須與以原摘要所組者相同(檔名以檔內solve.levelNumTradeBounds之級距設定組成，見optFnOf)
 *
 * @function
 * @param {Object} data 輸入原策略檔內容(含name、interval、mode、conds或keys、settings、summary)
 * @param {Object} settingsNew 輸入新口徑之settings
 * @param {Object} summaryNew 輸入新口徑之摘要
 * @returns {Array} 回傳不同之項目字串陣列(欄名, 檔名不同時另含'(fileName)')，空陣列代表不變
 * @example
 *
 * let data = { name: 'ETH', interval: '4hr', mode: 'long', keys: ['k1'], settings: { uIni: 1000 }, summary: { numTrade: 30, numTradeFin: 29, rWin: '60.00%' } }
 * console.log(checkInvariant(data, { uIni: 10000 }, { numTrade: 30, numTradeFin: 29, rWin: '60.00%' }), checkInvariant(data, { uIni: 10000 }, { numTrade: 3, numTradeFin: 29, rWin: '61.00%' }))
 * // => [] [ 'numTrade', 'rWin', '(fileName)' ]
 *
 */
let checkInvariant = (data, settingsNew, summaryNew) => {
    let keys = Array.isArray(data.keys) ? data.keys : map(data.conds, 'key')
    let bad = INVARIANT_KEYS.filter((k) => JSON.stringify(get(summaryNew, k)) !== JSON.stringify(get(data, `summary.${k}`)))
    let optFn = optFnOf(data)
    let fnNew = genStrategyFileName(data.name, data.interval, data.mode, keys, settingsNew, summaryNew, optFn)
    let fnOld = genStrategyFileName(data.name, data.interval, data.mode, keys, data.settings, data.summary, optFn)
    if (fnNew !== fnOld) {
        bad.push('(fileName)')
    }
    return bad
}


//numericSettingsOf, 檔內settings.uIni與settings.unit之數字字串視同同值之數字後供與目標比較(K4、K7); 不改呼叫端之物件
//  uIni: 回測套件(1.1.43起)對settings.uIni收數字字串並轉型(與settings.unit同規則: 給數字字串與給數字之訂單與摘要逐位相同), 故忠實重放可通過; 經isenum(gt 0)與cdbl(與回測套件同一判定與轉換)轉為數值, 非有效正數者原樣(必不等於目標)
//  unit: 經toNumericUnit〔不要求asOf與source〕轉為數值, 無效者原樣(必不等於目標)
//  檔內uIni或unit為無效值者, 忠實重放時回測套件即拒絕(reject, 訊息不含輸入值), 於core之catch記status 'error', 不會走到此處
let numericSettingsOf = (settings) => {
    let s = { ...settings }
    if (isenum(s.uIni, { gt: 0 })) {
        s.uIni = cdbl(s.uIni)
    }
    if (s.unit !== undefined && s.unit !== null) {
        try {
            s.unit = toNumericUnit(s.unit, 'unit')
        }
        catch (err) {}
    }
    return s
}

//unitsMaxOf, 全期最大同時持倉之單位數(同時持倉筆數numTradeAllMax × 每單之單位數), 摘要無numTradeAllMax時為null
let unitsMaxOf = (summary, settings) => {
    let n = get(summary, 'numTradeAllMax', null)
    return Number.isFinite(n) ? n * Number(get(settings, 'unit.units', 1)) : null //units可為整數字串(回測套件轉型)
}


/**
 * 建立策略庫重算之worker核心(每worker一次; 依求解時間窗快取取序列函數)
 *
 * 逐隻: ①讀策略檔(須有settings與solve之timeStart/timeEnd, 否則unsupported); ②忠實重放: 以檔內settings於求解時間窗重跑官方runStrategy, 摘要須與檔內逐欄相同(timeTest除外), 不同即mismatch; 有fitness欄而不等於現行calcFitness(settings,summary)者(舊版適應函數所算)另記fitnessStale(不擋重算);
 * ③verify模式至此為ok; dry/write模式: 檔內settings已等於目標(uIni與unit; 檔內settings.uIni與settings.unit之數字字串與同值之數字視為相同, 以轉為數值後比較, 回測套件1.1.43起對兩者皆收數字字串並轉型, 故數字字串之檔可忠實重放; settings.uIni為無效值〔非數字字串、非正數或非數值型別〕之檔於忠實重放時被回測套件拒絕, 例外即記error〔訊息不含輸入值〕、不中斷整批)且fitness為現行算法者為already〔逐位複製, 含原字串寫法〕, 否則以目標settings(檔內settings之uIni換為目標、unit設為目標)重跑, 交易筆數、已平倉筆數、勝率須與檔內相同且檔名不變(否則r8); 通過者為ok並附新內容(settings、summary、fitness換新, 其餘欄與鍵序不變); 單一策略之最大持倉名義不設門檻, 只回傳其值(uTradeAllMax1)與同時持倉之單位數(unitsMax1 = numTradeAllMax × units, 摘要無numTradeAllMax時為null)供統計
 * 檔名與檔內交易次數級距不一致之舊檔另記nameMismatch(只列、不擋; 輸出沿用原檔名); 例外一律以status 'error'回傳(不中斷worker)
 *
 * @function
 * @param {Object} init 輸入{fdIn,fdOhlc,fdParam,timeZone,mode,target}，mode為'verify'、'dry'或'write'，target為{uIni,unit}(verify時可為null; cfg.timeBasis有給時另含intervalSec與timeBasis, 新口徑之settings於末記之)
 * @param {Object} [deps={}] 輸入可替換之相依{runStrategy}(只供直接呼叫之測試注入; worker經workerData建立時不給)，預設{}
 * @returns {Function} 回傳async函數core(job) => 結果物件{fn,status,...}，job為{fn}
 * @example
 * need test in nodejs.
 *
 * let core = createCore({ fdIn, fdOhlc, fdParam, timeZone: 'Asia/Taipei', mode: 'dry', target: { uIni: 10000, unit } })
 * let r = await core({ fn: 'long_12_0123456789abcdef.json' })
 */
let createCore = (init, deps = {}) => {

    let run = isfun(get(deps, 'runStrategy', null)) ? deps.runStrategy : runStrategy
    let ott = makeOtt(init.timeZone)
    let provideData = dataProvide(init.fdOhlc, init.fdParam)
    let mode = init.mode
    let target = init.target

    //fgsOf, 依求解時間窗快取取序列函數(與estimKeys之取法相同)
    let fgsCache = new Map()
    let fgsOf = (ts, te) => {
        let k = `${ts}|${te}`
        let p = fgsCache.get(k)
        if (!p) {
            p = provideData.buildGetTimeSeriesByTimeRange(ts, te)
            fgsCache.set(k, p)
        }
        return p
    }

    let core = async (job) => {
        let fn = get(job, 'fn', '')
        try {

            //data
            let data = JSON.parse(fs.readFileSync(path.resolve(init.fdIn, fn), 'utf8'))
            let ts = get(data, 'solve.timeStart', '')
            let te = get(data, 'solve.timeEnd', '')
            if (!isobj(get(data, 'settings', null)) || !isobj(get(data, 'summary', null)) || !istime(ts) || !istime(te)) {
                return { fn, status: 'unsupported', detail: '缺settings、summary或solve.timeStart/timeEnd(非求解所落地之策略檔)' }
            }
            let fgs = await fgsOf(ts, te)
            let eqYr0 = p2r(get(data, 'summary.rEquivalentCumuProfitOrLossFinalNormYear', ''))
            let th = get(data, 'solve.thREquivalentCumuProfitOrLossFinalNormYear', null)

            //nm, 檔名與檔內交易次數級距不一致之舊檔(只列, 不擋; 輸出沿用原檔名)
            let keys = Array.isArray(data.keys) ? data.keys : map(data.conds, 'key')
            let nm = genStrategyFileName(data.name, data.interval, data.mode, keys, data.settings, data.summary, optFnOf(data)) !== fn ? { nameMismatch: true } : {}

            //忠實重放: 以檔內settings重跑, 摘要須與檔內逐欄相同
            let r0 = await run(ott, data, fgs)
            let diff0 = diffSummary(data.summary, r0.summary)
            if (diff0.length > 0) {
                return { fn, status: 'mismatch', diffFields: diff0, detail: diff0.slice(0, 5).map((k) => `${k}: ${JSON.stringify(data.summary[k])} → ${JSON.stringify(r0.summary[k])}`).join('; '), eqYr0, th, ...nm }
            }

            //fitnessStale, 檔內fitness與現行calcFitness(檔內settings,檔內summary)不同(舊版適應函數所算): 不擋重算(重算時以現行calcFitness重寫), 只記錄
            let ext = {}
            if (Object.prototype.hasOwnProperty.call(data, 'fitness')) {
                let f0 = calcFitness(data.settings, data.summary)
                if (data.fitness !== f0) {
                    ext = { fitnessStale: true, fitness0: data.fitness, fitnessNow: f0 }
                }
            }
            if (mode === 'verify') {
                return { fn, status: 'ok', eqYr0, th, ...ext, ...nm }
            }

            //already, 檔內settings已為目標口徑(且fitness為現行算法): uIni與unit之數字字串與同值之數字視為相同(以轉為數值後比較, unit之同一口徑即buildBaseUnitNotional之key相同, 記錄欄asOf、source與其他欄亦須相同)
            let settings1 = { ...data.settings, uIni: target.uIni, unit: target.unit }
            if (target.timeBasis !== undefined && target.timeBasis !== null) {
                settings1.intervalSec = target.intervalSec //時間口徑(目標口徑有之時; 見recomputeStrategies之cfg.timeBasis): 接於末, 檔內已有者原位覆寫
                settings1.timeBasis = target.timeBasis
            }
            if (isEqual(numericSettingsOf(data.settings), settings1) && !ext.fitnessStale) {
                return { fn, status: 'already', eqYr0, eqYr1: eqYr0, th, uTradeAllMax1: data.summary.uTradeAllMax, unitsMax1: unitsMaxOf(data.summary, data.settings), ...nm }
            }

            //新口徑重算
            let r1 = await run(ott, { ...data, settings: settings1 }, fgs)
            let sm1 = r1.summary
            let eqYr1 = p2r(get(sm1, 'rEquivalentCumuProfitOrLossFinalNormYear', ''))
            let bad = checkInvariant(data, settings1, sm1)
            if (bad.length > 0) {
                return { fn, status: 'r8', diffFields: bad, detail: bad.map((k) => `${k}: ${JSON.stringify(data.summary[k])} → ${JSON.stringify(sm1[k])}`).join('; '), eqYr0, eqYr1, th, ...nm }
            }

            //out, settings、summary、fitness換新, 其餘欄與鍵序不變
            let out = { ...data, settings: settings1, summary: sm1 }
            if (Object.prototype.hasOwnProperty.call(data, 'fitness')) {
                out.fitness = calcFitness(settings1, sm1)
            }
            let r = { fn, status: 'ok', diffFields: diffSummary(data.summary, sm1), eqYr0, eqYr1, th, uTradeAllMax1: sm1.uTradeAllMax, unitsMax1: unitsMaxOf(sm1, settings1), ...ext, ...nm }
            if (mode === 'write') {
                r.out = out
            }
            return r
        }
        catch (err) {
            return { fn, status: 'error', detail: String((err && err.message) || err) }
        }
    }

    return core
}


export { INVARIANT_KEYS, diffSummary, optFnOf, checkInvariant }
export default createCore
