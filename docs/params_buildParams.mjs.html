/**
 * 建立method名稱對應計算函數之映射(供w-data-trade dataConvert與dataCalibrate呼叫)
 *
 * 內建method為w-data-trade之各calc(取值類orig、ma、maNorm、ema、emaNorm、vec、vecNorm與指標類kdj、macd、vortex、rsi、klr、atr、bbpb、obv、mfi、adx、donchian、supertrend、keltner、vwap、cvd、ats、ichimoku、candle、dirChange)；dirChange(方向改變法calcDirChange)之表為門檻百分比(期別段`${門檻}pct`)，intervalSec只檢核、不換算其標籤
 * 每次呼叫皆把intervalSec(每根K線之秒數)併入傳給calc之opt(w-data-trade >= 1.0.50，根數為本)：各calc內建週期表之值為根數，不隨K線週期改變，期別標籤(即輸出key之period)= 根數 × intervalSec(4hr為14400即與內建標籤相同)；呼叫端給之optMethod(index側偏移參數等)原樣保留，其內之intervalSec可覆寫
 * calcs可擴充或覆寫method，函數簽章(arr, keyIn, optMethod) => rrs，rrs為各期[{period, len, vs: [{time, ...keyOuts}]}]
 * intervalSec依w-data-trade之chkIntervalSec檢核(60之正整數倍且至多31536000，數值或數字字串，轉為數值)，給了而無效即throw `invalid opt.intervalSec: must be a positive integer multiple of 60 up to 31536000`，null與未給取預設14400；已移除之舊選項barsPerDay視為未知鍵，不收
 *
 * @function
 * @param {Object} [calcs={}] 輸入自訂計算函數物件
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Number|String} [opt.intervalSec=14400] 輸入每根K線之秒數，60之正整數倍且至多31536000(數值或數字字串)，預設14400(4hr)
 * @returns {Object} 回傳method對應函數之物件
 * @example
 *
 * let kp = buildKp({ myCalc: async (arr, keyIn, optMethod) => [{ period: '1bar', len: 1, vs: [] }] }, { intervalSec: 3600 })
 * console.log(Object.keys(kp).length, typeof kp.rsi, typeof kp.myCalc)
 * // => 27 function function
 *
 */
let buildKp = (calcs = {}, opt = {}) => {
    let intervalSecOpt = get(opt, 'intervalSec', null)
    let intervalSec = (intervalSecOpt === null || intervalSecOpt === undefined) ? INTERVAL_SEC_BASE : chkIntervalSec(intervalSecOpt) //給了即檢核(無效即throw), 檢測之值即使用之值
    let o = (opt2 = {}, ext = {}) => ({ intervalSec, ...opt2, ...ext })
    let kp = {
        orig: async (arr, key, opt2 = {}) => wdt.calcOrig(arr, key, o(opt2)),
        ma: (arr, key, opt2 = {}) => wdt.calcMa(arr, key, o(opt2, { norm: false })),
        maNorm: (arr, key, opt2 = {}) => wdt.calcMa(arr, key, o(opt2, { norm: true })),
        ema: (arr, key, opt2 = {}) => wdt.calcEma(arr, key, o(opt2, { norm: false })),
        emaNorm: (arr, key, opt2 = {}) => wdt.calcEma(arr, key, o(opt2, { norm: true })),
        vec: (arr, key, opt2 = {}) => wdt.calcVec(arr, key, o(opt2, { norm: false })),
        vecNorm: (arr, key, opt2 = {}) => wdt.calcVec(arr, key, o(opt2, { norm: true })),
        kdj: (arr, key, opt2 = {}) => wdt.calcKdj(arr, key, o(opt2)),
        macd: (arr, key, opt2 = {}) => wdt.calcMacd(arr, key, o(opt2)),
        vortex: (arr, key, opt2 = {}) => wdt.calcVortex(arr, key, o(opt2)),
        rsi: (arr, key, opt2 = {}) => wdt.calcRsi(arr, key, o(opt2)),
        klr: (arr, key, opt2 = {}) => wdt.calcKlr(arr, key, o(opt2)),
        atr: (arr, key, opt2 = {}) => wdt.calcAtr(arr, key, o(opt2)),
        bbpb: (arr, key, opt2 = {}) => wdt.calcBbpb(arr, key, o(opt2)),
        obv: (arr, key, opt2 = {}) => wdt.calcObv(arr, key, o(opt2)),
        mfi: (arr, key, opt2 = {}) => wdt.calcMfi(arr, key, o(opt2)),
        adx: (arr, key, opt2 = {}) => wdt.calcAdx(arr, key, o(opt2)),
        donchian: (arr, key, opt2 = {}) => wdt.calcDonchian(arr, key, o(opt2)),
        supertrend: (arr, key, opt2 = {}) => wdt.calcSupertrend(arr, key, o(opt2)),
        keltner: (arr, key, opt2 = {}) => wdt.calcKeltner(arr, key, o(opt2)),
        vwap: (arr, key, opt2 = {}) => wdt.calcVwap(arr, key, o(opt2)),
        cvd: (arr, key, opt2 = {}) => wdt.calcCvd(arr, key, o(opt2)),
        ats: (arr, key, opt2 = {}) => wdt.calcAts(arr, key, o(opt2)),
        ichimoku: (arr, key, opt2 = {}) => wdt.calcIchimoku(arr, key, o(opt2)),
        candle: (arr, key, opt2 = {}) => wdt.calcCandle(arr, key, o(opt2)),
        dirChange: (arr, key, opt2 = {}) => wdt.calcDirChange(arr, key, o(opt2)),
    }
    if (iseobj(calcs)) {
        for (let k of Object.keys(calcs)) {
            if (!isfun(calcs[k])) {
                throw new Error(`opt.calcs.${k}須為函數(arr, keyIn, optMethod) => rrs`)
            }
            kp[k] = (arr, key, opt2 = {}) => calcs[k](arr, key, o(opt2))
        }
    }
    return kp
}


/**
 * 檢核參數定義並正規化(keysIn空時為['none']；未指定keyIn之deal對每個keysIn各展開一筆，使nm能逐keyIn凍結)
 *
 * @function
 * @param {Object} def 輸入參數定義物件(格式見catalog.mjs檔頭)
 * @param {Number} i 輸入於定義陣列之索引(僅用於錯誤訊息)
 * @returns {Object} 回傳正規化後之定義{id,cointpe,keyKind,fields,keysIn,funAddKeysIn,methods}
 * @example
 *
 * let def = validateDef({ id: 'price_ohlc_close', cointpe: 'price', keyKind: 'ohlc', fields: ['Close'], keysIn: ['Close'], methods: [{ method: 'maNorm', deals: [{ keyOut: 'param' }] }] }, 0)
 * console.log(JSON.stringify(def.methods[0].deals))
 * // => [{"keyIn":"Close","keyOut":"param","nm":null}]
 *
 */
let validateDef = (def, i) => {
    if (!iseobj(def)) {
        throw new Error(`defs[${i}]須為物件`)
    }
    let id = get(def, 'id', '')
    if (!isestr(id)) {
        throw new Error(`defs[${i}].id須為非空字串`)
    }
    let cointpe = get(def, 'cointpe', '')
    if (cointpe !== 'price' && cointpe !== 'index') {
        throw new Error(`defs[${id}].cointpe[${cointpe}]須為'price'或'index'`)
    }
    let keyKind = get(def, 'keyKind', '')
    if (!isestr(keyKind)) {
        throw new Error(`defs[${id}].keyKind須為非空字串`)
    }
    let keysIn = get(def, 'keysIn', [])
    if (!Array.isArray(keysIn)) {
        throw new Error(`defs[${id}].keysIn須為陣列`)
    }
    let keysInUse = keysIn.length > 0 ? keysIn : ['none']
    let methods = get(def, 'methods', null)
    if (!isearr(methods)) {
        throw new Error(`defs[${id}].methods須為非空陣列`)
    }
    let fields = get(def, 'fields', [])
    if (!Array.isArray(fields)) {
        throw new Error(`defs[${id}].fields須為陣列`)
    }
    let ms = map(methods, (m, j) => {
        if (!iseobj(m) || !isestr(m.method)) {
            throw new Error(`defs[${id}].methods[${j}].method須為非空字串`)
        }
        if (!isearr(m.deals)) {
            throw new Error(`defs[${id}].methods[${j}].deals須為非空陣列`)
        }
        //deals, 未指定keyIn者對每個keysIn各展開一筆(使nm能逐keyIn凍結)
        let deals = []
        for (let d of m.deals) {
            if (!iseobj(d) || !isestr(d.keyOut)) {
                throw new Error(`defs[${id}].methods[${j}].deals之各元素須含keyOut`)
            }
            if (isestr(d.keyIn)) {
                deals.push({ keyIn: d.keyIn, keyOut: d.keyOut, nm: iseobj(d.nm) ? d.nm : null })
            }
            else {
                for (let keyIn of keysInUse) {
                    deals.push({ keyIn, keyOut: d.keyOut, nm: iseobj(d.nm) ? d.nm : null })
                }
            }
        }
        return {
            method: m.method,
            optMethod: iseobj(m.optMethod) ? { ...m.optMethod } : null,
            deals,
        }
    })
    return {
        id,
        cointpe,
        keyKind,
        fields,
        keysIn: keysInUse,
        funAddKeysIn: isfun(def.funAddKeysIn) ? def.funAddKeysIn : null,
        methods: ms,
    }
}


/**
 * 取K線陣列前段(至多50筆)各筆皆有數值之欄位集合(供判定定義所需欄位是否齊備)
 *
 * @function
 * @param {Array} arr 輸入K線陣列
 * @returns {Set} 回傳欄位名稱集合
 * @example
 *
 * console.log([...fieldsOf([{ time: 't1', Close: 1, Volumn: 10 }, { time: 't2', Close: 2, Volumn: null }])].join(','))
 * // => Close
 *
 */
let fieldsOf = (arr) => {
    let n = Math.min(arr.length, 50)
    let cnt = {}
    for (let i = 0; i < n; i++) {
        for (let k of Object.keys(arr[i])) {
            if (isnum(arr[i][k])) {
                cnt[k] = (cnt[k] || 0) + 1
            }
        }
    }
    let s = new Set()
    for (let k of Object.keys(cnt)) {
        if (cnt[k] === n) {
            s.add(k)
        }
    }
    return s
}


/**
 * 取時間區間內之K線(含頭含尾)；p為null時回傳原陣列
 *
 * @function
 * @param {Array} arr 輸入K線陣列
 * @param {Object} p 輸入期間{timeStart,timeEnd}或null
 * @returns {Array} 回傳過濾後之陣列
 * @example
 *
 * console.log(sliceByPeriod([{ time: '2023-01-01T00:00:00' }, { time: '2023-01-02T00:00:00' }], { timeStart: '2023-01-01T12:00:00', timeEnd: '2023-01-03T00:00:00' }).length)
 * // => 1
 *
 */
let sliceByPeriod = (arr, p) => {
    if (!p) {
        return arr
    }
    return filter(arr, (v) => v.time >= p.timeStart && v.time <= p.timeEnd)
}


/**
 * 正規化統計量(nm)之凍結鍵(不含期別: 各period共用同一組nm; opt.nmScope為'period'時於nm內以byPeriod另列各期別之值)
 *
 * @function
 * @param {Object} def 輸入參數定義(取其id)
 * @param {String} method 輸入method名稱
 * @param {String} keyIn 輸入來源欄位名稱('none'代表無)
 * @param {String} keyOut 輸入輸出欄位名稱
 * @returns {String} 回傳鍵字串`${id}|${method}|${keyIn}|${keyOut}`
 * @example
 *
 * console.log(nmKeyOf({ id: 'price_ohlc_rsi' }, 'rsi', 'none', 'RSI'))
 * // => price_ohlc_rsi|rsi|none|RSI
 *
 */
let nmKeyOf = (def, method, keyIn, keyOut) => `${def.id}|${method}|${keyIn}|${keyOut}`


/**
 * 決定計算nm之期間：opt.nmPeriod為'all'時用全部數據(回null)、為物件時用其{timeStart,timeEnd}，預設為cfg.train(無則全部)
 *
 * @function
 * @param {Object} cfg 輸入normalizeCfg正規化後之設定物件
 * @param {Object} opt 輸入buildParams之設定物件
 * @returns {Object|null} 回傳{timeStart,timeEnd}或null
 * @example
 *
 * let cfg = { train: { timeStart: '2022-07-01T00:00:00', timeEnd: '2025-10-06T00:00:00' } }
 * console.log(JSON.stringify(resolveNmPeriod(cfg, {})), resolveNmPeriod(cfg, { nmPeriod: 'all' }), JSON.stringify(resolveNmPeriod({ train: null }, {})))
 * // => {"timeStart":"2022-07-01T00:00:00","timeEnd":"2025-10-06T00:00:00"} null null
 *
 */
let resolveNmPeriod = (cfg, opt) => {
    let p = get(opt, 'nmPeriod', null)
    if (p === 'all') {
        return null
    }
    if (iseobj(p)) {
        if (!istime(p.timeStart) || !istime(p.timeEnd) || !(p.timeStart < p.timeEnd)) {
            throw new Error(`invalid opt.nmPeriod, 須為{timeStart,timeEnd}且timeStart早於timeEnd`)
        }
        return { timeStart: p.timeStart, timeEnd: p.timeEnd }
    }
    if (cfg.train) {
        return { timeStart: cfg.train.timeStart, timeEnd: cfg.train.timeEnd }
    }
    return null
}


//NM_SCOPES, NM_STATS, nm之取法選項: nmScope 'key'為同一輸出欄各期別共用一組(預設)、'period'為各期別各自一組; nmStat 'meanStd'為平均與標準差(預設)、'medianIqr'為中位數與四分位距
let NM_SCOPES = ['key', 'period']
let NM_STATS = ['meanStd', 'medianIqr']

//NM_IQR_DIV, 四分位距換算為標準差之除數(常態分布下 (q75 − q25) ÷ 1.349 ≈ σ)
let NM_IQR_DIV = 1.349

//NM_MIN_N, opt.nmScope為'period'時byPeriod收錄一個期別所需之最少筆數(該期別於nm期間內之有效值個數n): 固定值, 不設選項; 少於此者不收而沿用鍵層之值
//  依據(實測於合成之重尾與波動叢聚K線、2個亂數種子、內建目錄全部140個輸出; 以不同大小之nm窗估各期別之尺度, 與其後長窗之同期別尺度之誤差|ln(估計/其後長窗)|, 比較「自身之估計」與「沿用鍵層之攤平尺度」; 標準差與四分位距兩種尺度):
//    n 2 ~ 14: 自身之誤差中位數1.0 ~ 2.1, 鍵層0.3 ~ 1.2(自身明顯較差); n 15 ~ 19: 自身較差(0.51 ~ 0.76對0.37 ~ 0.63); n 20 ~ 29: 兩者相當(中位數0.34 ~ 0.39對0.34 ~ 0.42, 逐格勝出之比例0.45 ~ 0.46); n 30 ~ 49: 自身之中位數略小(0.27 ~ 0.34對0.30 ~ 0.41)而第90百分位數不小於鍵層、逐格勝出之比例0.47 ~ 0.49(不過半); n 50 ~ 99: 中位數(0.19 ~ 0.24對0.28 ~ 0.34)、第90百分位數、逐格勝出之比例(0.54 ~ 0.57)三者皆為自身較佳(兩種尺度、兩個種子皆然)
//  故取50(三項指標皆不遜於鍵層之最小分組下界); 內建目錄於一般之nm期間(數百根以上)下各期別之n遠大於此, 下限只在nm期間短(例: 不足約230根而含180根之期別)時才起作用
let NM_MIN_N = 50


//nmMedianIqrOf, 以中位數為中心、(q75 − q25) ÷ 1.349為尺度取{avg,std}; 中心或尺度非有限、尺度不大於0者回null
let nmMedianIqrOf = (st) => {
    let s = (st.q75 - st.q25) / NM_IQR_DIV
    if (Number.isFinite(st.median) && Number.isFinite(s) && s > 0) {
        return { avg: st.median, std: s }
    }
    return null
}


/**
 * 檢核並取得opt.nmScope與opt.nmStat：未給(null或undefined)者回null，給了而無效即throw固定訊息(不含輸入值)
 *
 * @function
 * @param {Object} opt 輸入buildParams之設定物件
 * @returns {Object} 回傳{nmScope,nmStat}，未給者為null
 * @example
 *
 * console.log(JSON.stringify(resolveNmOpt({})), JSON.stringify(resolveNmOpt({ nmScope: 'period', nmStat: 'medianIqr' })))
 * // => {"nmScope":null,"nmStat":null} {"nmScope":"period","nmStat":"medianIqr"}
 *
 */
let resolveNmOpt = (opt) => {
    let nmScope = get(opt, 'nmScope', null)
    let nmStat = get(opt, 'nmStat', null)
    if (nmScope !== null && !includes(NM_SCOPES, nmScope)) {
        throw new Error(`invalid opt.nmScope: must be one of 'key', 'period'`)
    }
    if (nmStat !== null && !includes(NM_STATS, nmStat)) {
        throw new Error(`invalid opt.nmStat: must be one of 'meanStd', 'medianIqr'`)
    }
    return { nmScope, nmStat }
}


/**
 * 由dataConvert回傳之一組統計{avg,std,median,q25,q75}取正規化常數{avg,std}：nmStat為'meanStd'取平均與標準差，為'medianIqr'取中位數與(q75 − q25) ÷ 1.349；尺度非有限或不大於0者'medianIqr'先退回平均與標準差，仍不行即回null
 *
 * @function
 * @param {Object} st 輸入統計物件{avg,std,median,q25,q75}
 * @param {String} nmStat 輸入取法，'meanStd'或'medianIqr'
 * @returns {Object|null} 回傳{avg,std}，無法取得有效常數時回null
 * @example
 *
 * let st = { avg: 10, std: 2, median: 8, q25: 7, q75: 9.349 }
 * console.log(JSON.stringify(nmOfStat(st, 'meanStd')), JSON.stringify(nmOfStat(st, 'medianIqr')), JSON.stringify(nmOfStat({ ...st, q75: 7 }, 'medianIqr')), nmOfStat({ ...st, std: 0 }, 'meanStd'))
 * // => {"avg":10,"std":2} {"avg":8,"std":1.741289844329133} {"avg":10,"std":2} null
 *
 */
let nmOfStat = (st, nmStat) => {
    let nm = nmStat === 'medianIqr' ? nmMedianIqrOf(st) : null
    if (nm) {
        return nm
    }
    if (Number.isFinite(st.avg) && Number.isFinite(st.std) && st.std > 0) {
        return { avg: st.avg, std: st.std }
    }
    return null
}


/**
 * 由dataConvert回傳之各期統計(periods)取各期別各自之正規化常數：只收n(該期別之有效值個數)至少50(固定值，見NM_MIN_N)且尺度大於0之期別(取法同nmOfStat)，其餘期別不列(使用時沿用鍵層之常數)
 *
 * @function
 * @param {Array} periods 輸入各期統計陣列[{period,n,avg,std,median,q25,q75}]
 * @param {String} nmStat 輸入取法，'meanStd'或'medianIqr'
 * @returns {Object} 回傳{期別標籤:{avg,std}}
 * @example
 *
 * let periods = [{ period: '1day', n: 50, avg: 1, std: 2, median: 0, q25: -1, q75: 1 }, { period: '2day', n: 1, avg: 3, std: null, median: 3, q25: 3, q75: 3 }, { period: '4day', n: 40, avg: 5, std: 0, median: 5, q25: 5, q75: 5 }, { period: '8day', n: 49, avg: 2, std: 3, median: 2, q25: 0, q75: 4 }, { period: '16day', n: 50, avg: 7, std: 3, median: 7, q25: 5.5, q75: 8.5 }]
 * console.log(JSON.stringify(nmByPeriodOf(periods, 'meanStd')), JSON.stringify(nmByPeriodOf(periods, 'medianIqr')))
 * // => {"1day":{"avg":1,"std":2},"16day":{"avg":7,"std":3}} {"1day":{"avg":0,"std":1.4825796886582654},"16day":{"avg":7,"std":2.2238695329873983}}
 *
 */
let nmByPeriodOf = (periods, nmStat) => {
    let byPeriod = {}
    for (let p of periods) {
        if (!(typeof p.n === 'number' && p.n >= NM_MIN_N)) {
            continue
        }
        let nm = nmOfStat(p, nmStat)
        if (nm) {
            byPeriod[p.period] = nm
        }
    }
    return byPeriod
}


/**
 * 由K線序列計算指標參數(因子)序列並儲存至cfg.fds.param
 *
 * 參數定義預設為內建目錄(抽自使用端既有管線之34個定義，另增方向改變法，共35個定義、140個輸出)，可由opt.defs整組替換或opt.extraDefs追加，定義格式見src/params/catalog.mjs
 * 各定義經三道過濾: cfg.useIndex為false時剔除cointpe為'index'者；數據缺少定義fields任一欄位者剔除(供僅有OHLC之股價等數據)；opt.only/opt.exclude依id取捨
 * z-score正規化之nm(avg,std)依opt.nmPeriod(預設訓練期)由數據自算並凍結於cfg.fds.meta/params.json，之後重算參數時沿用凍結值(數據延長不致使既有策略之門檻失真)；opt.recalibrate為true時重算
 * nm之取法有二軸(皆選配，預設即現行): opt.nmScope為'key'(預設)時同一輸出欄(`${id}|${method}|${keyIn}|${keyOut}`)之各期別共用一組nm，為'period'時另依各期別各自算一組(meta.nm[鍵] = {avg,std,byPeriod:{期別標籤:{avg,std}}}，頂層為各期別攤平之值，供無對應期別時沿用；byPeriod只收有效值至少50筆(固定下限，不設選項)且尺度大於0之期別，其餘期別沿用頂層；於本套件外自行以meta.nm正規化新數據者須依期別取byPeriod，只讀頂層會與寫出之序列不一致)；opt.nmStat為'meanStd'(預設)時取平均與標準差，為'medianIqr'時取中位數與(q75 − q25) ÷ 1.349(尺度不可用者先退回平均與標準差，仍不行為{avg:0,std:1})，後者對重尾較穩健
 * 給了而無效即throw `invalid opt.nmScope: must be one of 'key', 'period'`、`invalid opt.nmStat: must be one of 'meanStd', 'medianIqr'`(不含輸入值，先於任何建夾與寫檔，含口徑標記)；二者於新建或opt.recalibrate重算時凍結於meta/params.json之nmScope與nmStat(兩者皆為預設時不寫入，使既有工作根逐位元組不變；meta無此二欄者視為預設)，已凍結者沿用凍結值(本次所給不同者log一行，須opt.recalibrate才改；meta內凍結之nmScope或nmStat無效者throw固定訊息`invalid cfg.root: must have a valid nmScope in meta/params.json, one of 'key', 'period'; use opt.recalibrate or a new work root`、`invalid cfg.root: must have a valid nmStat in meta/params.json, one of 'meanStd', 'medianIqr'; use opt.recalibrate or a new work root`)；二者之一非預設而所裝w-data-trade之dataConvert未回傳periods、median、q25、q75者throw(須較新之w-data-trade)
 * index側帶optMethod之指標預設以w-data-trade dataCalibrate對price側同名keyOut校正偏移參數(新幣種之index量級與ETH不同)，結果亦凍結於meta；opt.calibrateIndex為false時直接採定義值
 * opt.series可直接寫入外部自行計算之序列({key:[{time,param}]})，opt.calcs可擴充或覆寫method對應之計算函數
 * 輸出檔名由w-data-trade dataConvert決定: `${cointpe}_${keyKind}_${keyIn}_${keyOut}_${method}_${period}.json`
 * 每根K線之秒數一律為cfg.intervalSec: 交給dataConvert之各method之optMethod.intervalSec皆為cfg.intervalSec(含不讀選項之orig; 定義之optMethod內若有intervalSec亦以cfg.intervalSec為準)，dataCalibrate之搜尋參數不含intervalSec(optMethodInit與optMethodRef皆去除intervalSec，由buildKp注入)，故凍結於meta之optMethod不含intervalSec
 * 工作根之口徑(meta/basis.json，見src/util/basis.mjs)於任何寫檔之前檢查: 非4hr之工作根須有一致之口徑標記(首次建立時寫入)，無標記而已有因子序列或meta/params.json者(例如以依時間換算根數之舊版建立者)即throw；4hr(直接下載或外部數據)且未給timeBasis者不寫標記(標記存在時照常讀取比對)
 *
 * @function
 * @param {Object} cfg 輸入normalizeCfg正規化後之設定物件
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Array} [opt.defs] 輸入參數定義陣列，未給時用內建目錄
 * @param {Array} [opt.extraDefs=[]] 輸入追加之參數定義陣列
 * @param {Array} [opt.only] 輸入只計算之定義id陣列
 * @param {Array} [opt.exclude] 輸入排除之定義id陣列
 * @param {Object} [opt.calcs] 輸入method對應之自訂計算函數物件，簽章(arr, keyIn, optMethod) => rrs(各期{period,len,vs})
 * @param {Object} [opt.series] 輸入外部序列物件{key:[{time,param}]}(值亦可為async (cfg, key) => 陣列之函數)，直接寫入param資料夾
 * @param {String|Object} [opt.nmPeriod] 輸入計算nm之期間，'all'為全部數據，物件為{timeStart,timeEnd}，預設cfg.train
 * @param {String} [opt.nmScope='key'] 輸入nm之範圍，'key'為同一輸出欄之各期別共用一組，'period'為各期別各自一組(凍結於meta，見上)，預設'key'
 * @param {String} [opt.nmStat='meanStd'] 輸入nm之統計取法，'meanStd'為平均與標準差，'medianIqr'為中位數與(q75 − q25) ÷ 1.349(凍結於meta，見上)，預設'meanStd'
 * @param {Boolean} [opt.calibrateIndex=true] 輸入是否校正index側optMethod布林值
 * @param {Boolean} [opt.recalibrate=false] 輸入是否捨棄meta內凍結之nm與optMethod重算布林值
 * @param {Boolean} [opt.useShowLog=false] 輸入是否顯示log布林值
 * @returns {Promise} 回傳Promise，resolve為{nKeys,keys,defs,dropped,calibrated,nmComputed,meta,nmScope,nmStat}，nmScope與nmStat為本次實際採用(新建取opt、已凍結者為凍結值)之取法
 * @example
 * need test in nodejs.
 *
 * let r = await buildParams(cfg, { only: ['price_ohlc_close', 'price_ohlc_rsi'] })
 * console.log(r.nKeys, r.dropped.length)
 */
let buildParams = async (cfg, opt = {}) => {

    let useShowLog = get(opt, 'useShowLog', false) === true
    let log = (...args) => {
        if (useShowLog) {
            console.log(...args)
        }
    }

    //nmScopeOpt, nmStatOpt, 選項檢核(先於任何建夾與寫檔; 給了而無效即throw, 未給為null)
    let { nmScope: nmScopeOpt, nmStat: nmStatOpt } = resolveNmOpt(opt)

    //basisWrite, 工作根之口徑檢查(先於任何建夾與寫檔; 不一致或無標記而已有產物即throw), 須寫標記者於計算前寫入
    let basisWrite = checkBasis(cfg, 'params')

    //fds
    let fdOhlc = cfg.fds.ohlc
    let fdParam = ensureFolder(cfg.fds.param)
    let fdMeta = ensureFolder(cfg.fds.meta)

    //arrP, arrI
    let arrP = readJsonOr(path.resolve(fdOhlc, `${cfg.keyOhlc}.json`), null)
    if (!isearr(arrP)) {
        throw new Error(`尚無K線序列[${cfg.keyOhlc}], 請先執行buildOhlc`)
    }
    let arrI = null
    if (cfg.useIndex) {
        arrI = readJsonOr(path.resolve(fdOhlc, `${cfg.keyIndex}.json`), null)
        if (!isearr(arrI)) {
            throw new Error(`尚無溢價指數序列[${cfg.keyIndex}], 請先執行buildOhlc`)
        }
    }

    //defs
    let defsIn = isearr(get(opt, 'defs')) ? opt.defs : CATALOG
    let extra = get(opt, 'extraDefs', [])
    if (!Array.isArray(extra)) {
        throw new Error(`opt.extraDefs須為陣列`)
    }
    let defs = map([...defsIn, ...extra], validateDef)

    //filter
    let dropped = []
    let fieldsP = fieldsOf(arrP)
    let fieldsI = arrI ? fieldsOf(arrI) : new Set()
    let only = get(opt, 'only', null)
    let exclude = get(opt, 'exclude', null)
    defs = filter(defs, (def) => {
        if (isearr(only) && !includes(only, def.id)) {
            dropped.push({ id: def.id, reason: '不在opt.only' })
            return false
        }
        if (isearr(exclude) && includes(exclude, def.id)) {
            dropped.push({ id: def.id, reason: '列於opt.exclude' })
            return false
        }
        if (def.cointpe === 'index' && !cfg.useIndex) {
            dropped.push({ id: def.id, reason: 'useIndex為false' })
            return false
        }
        let fields = def.cointpe === 'index' ? fieldsI : fieldsP
        let miss = filter(def.fields, (f) => !fields.has(f))
        if (miss.length > 0) {
            dropped.push({ id: def.id, reason: `數據缺欄位[${miss.join(',')}]` })
            return false
        }
        return true
    })
    if (defs.length === 0 && !iseobj(get(opt, 'series'))) {
        throw new Error(`無可計算之參數定義(dropped: ${JSON.stringify(dropped)})`)
    }

    //kp, 每根K線之秒數(cfg.intervalSec, 4hr為14400)併入每次calc呼叫(根數為本): 各calc週期表之根數不變, 期別標籤(輸出key之period) = 根數 × 每根秒數
    let kp = buildKp(get(opt, 'calcs', {}), { intervalSec: cfg.intervalSec })
    for (let def of defs) {
        for (let m of def.methods) {
            if (!haskey(kp, m.method)) {
                throw new Error(`defs[${def.id}].method[${m.method}]無對應之計算函數, 請由opt.calcs提供`)
            }
        }
    }

    //basis, 非預設口徑之工作根首次建立時寫入口徑標記(檢查已於函數開頭完成; 寫入於任何序列檔之前, 以重讀 → 比對 → 寫入防他行程同時寫入不同口徑)
    commitBasis(cfg, basisWrite)

    //omitIv, 去除optMethod內之intervalSec(每根秒數一律由buildKp注入cfg.intervalSec, 不作dataCalibrate之搜尋參數, 亦不凍結於meta)
    let omitIv = (om) => omit(om, ['intervalSec'])

    //withIv, 交給dataConvert之optMethod一律帶cfg.intervalSec(各method同一每根秒數, 含不讀選項之orig)
    let withIv = (om) => ({ ...om, intervalSec: cfg.intervalSec })

    //meta
    let fpMeta = path.resolve(fdMeta, 'params.json')
    let meta = readJsonOr(fpMeta, null)
    let metaNew = false
    if (!iseobj(meta) || get(opt, 'recalibrate', false) === true) {
        meta = { nm: {}, optMethod: {}, nmPeriod: null }
        metaNew = true
    }
    if (!iseobj(meta.nm)) {
        meta.nm = {}
    }
    if (!iseobj(meta.optMethod)) {
        meta.optMethod = {}
    }

    //nmScope, nmStat, 新建或recalibrate重算時取opt之值(未給取預設), 已凍結者沿用凍結之值(meta無此二欄者視為預設), 須opt.recalibrate才改
    let nmScope = nmScopeOpt === null ? 'key' : nmScopeOpt
    let nmStat = nmStatOpt === null ? 'meanStd' : nmStatOpt
    if (!metaNew) {
        nmScope = get(meta, 'nmScope', 'key')
        nmStat = get(meta, 'nmStat', 'meanStd')
        if (!includes(NM_SCOPES, nmScope)) {
            throw new Error(`invalid cfg.root: must have a valid nmScope in meta/params.json, one of 'key', 'period'; use opt.recalibrate or a new work root`)
        }
        if (!includes(NM_STATS, nmStat)) {
            throw new Error(`invalid cfg.root: must have a valid nmStat in meta/params.json, one of 'meanStd', 'medianIqr'; use opt.recalibrate or a new work root`)
        }
        if ((nmScopeOpt !== null && nmScopeOpt !== nmScope) || (nmStatOpt !== null && nmStatOpt !== nmStat)) {
            log(`沿用meta凍結之nmScope[${nmScope}]與nmStat[${nmStat}](本次指定[${nmScopeOpt === null ? '未給' : nmScopeOpt}]與[${nmStatOpt === null ? '未給' : nmStatOpt}]未採, 須opt.recalibrate才重算)`)
        }
    }
    let nmUseNew = nmScope !== 'key' || nmStat !== 'meanStd'
    let nmPeriodWant = resolveNmPeriod(cfg, opt)
    let nmPeriod = meta.nmPeriod ? meta.nmPeriod : nmPeriodWant
    if (meta.nmPeriod && nmPeriodWant && (meta.nmPeriod.timeStart !== nmPeriodWant.timeStart || meta.nmPeriod.timeEnd !== nmPeriodWant.timeEnd)) {
        log(`沿用meta凍結之nm期間[${meta.nmPeriod.timeStart}~${meta.nmPeriod.timeEnd}](本次指定[${nmPeriodWant.timeStart}~${nmPeriodWant.timeEnd}]未採, 須opt.recalibrate才重算)`)
    }
    let arrPN = sliceByPeriod(arrP, nmPeriod)
    let arrIN = arrI ? sliceByPeriod(arrI, nmPeriod) : null
    if (arrPN.length === 0 || (arrIN && arrIN.length === 0)) {
        throw new Error(`nm期間[${JSON.stringify(nmPeriod)}]內無K線數據`)
    }

    //Pass A: index側optMethod校正(nm依賴optMethod, 故先做)
    let calibrated = []
    let calibrateIndex = get(opt, 'calibrateIndex', true) !== false
    for (let def of defs) {
        if (def.cointpe !== 'index') {
            continue
        }
        for (let m of def.methods) {
            if (!iseobj(m.optMethod)) {
                continue
            }
            let mk = `${def.id}:${m.method}`
            if (haskey(meta.optMethod, mk)) {
                continue
            }
            let defRef = find(defs, (d) => d.cointpe === 'price' && d.keyKind === def.keyKind && find(d.methods, { method: m.method }))
            if (!defRef) {
                defRef = find(map(CATALOG, validateDef), (d) => d.cointpe === 'price' && d.keyKind === def.keyKind && find(d.methods, { method: m.method }))
            }
            let optMethodInit = omitIv(m.optMethod)
            if (!calibrateIndex || !defRef) {
                meta.optMethod[mk] = optMethodInit
                continue
            }
            let mRef = find(defRef.methods, { method: m.method })
            let outsRef = new Set(map(mRef.deals, 'keyOut'))
            let pairs = []
            for (let d of m.deals) {
                let keyOutRef = d.keyOut.replace(/Mod$/, '')
                if (outsRef.has(keyOutRef)) {
                    pairs.push({ keyOutSrc: d.keyOut, keyOutRef })
                }
            }
            if (pairs.length === 0) {
                meta.optMethod[mk] = optMethodInit
                continue
            }
            let keyIn = def.keysIn[0]
            log(`校正index側optMethod: ${mk} (pairs ${map(pairs, 'keyOutSrc').join(',')})`)
            let r = await wdt.dataCalibrate(kp[m.method], arrIN, arrPN, pairs, optMethodInit, {
                keyIn,
                params: Object.keys(optMethodInit), //搜尋參數明列為optMethod之各鍵(不含intervalSec; 與dataCalibrate之預設相同)
                optMethodRef: iseobj(mRef.optMethod) ? omitIv(mRef.optMethod) : {},
                funLog: useShowLog ? (msg) => console.log(`  ${msg}`) : null,
            })
            meta.optMethod[mk] = r.optMethod
            calibrated.push({ id: def.id, method: m.method, adopt: r.adopt, optMethod: r.optMethod, optMethodInit, score0: r.score0, scoreBest: r.scoreBest, nEval: r.nEval })
        }
    }

    //Pass B: nm(缺者由nm期間數據自算)與寫出序列
    let nmComputed = []
    for (let def of defs) {
        let arrSrc = def.cointpe === 'index' ? arrI : arrP
        let arrNm = def.cointpe === 'index' ? arrIN : arrPN
        let methodsRaw = map(def.methods, (m) => {
            let om = get(meta.optMethod, `${def.id}:${m.method}`, null)
            return {
                method: m.method,
                optMethod: iseobj(om) ? om : (iseobj(m.optMethod) ? m.optMethod : {}),
                deals: m.deals,
            }
        })

        //need, 定義內有無nm尚未凍結之deal(定義自帶nm者直接採用並凍結)
        let need = false
        for (let m of methodsRaw) {
            for (let d of m.deals) {
                let nk = nmKeyOf(def, m.method, d.keyIn, d.keyOut)
                if (haskey(meta.nm, nk)) {
                    continue
                }
                if (iseobj(d.nm) && isnum(d.nm.avg) && isnum(d.nm.std)) {
                    meta.nm[nk] = { avg: d.nm.avg, std: d.nm.std }
                    if (d.nm.byPeriod !== undefined && d.nm.byPeriod !== null) {
                        meta.nm[nk].byPeriod = d.nm.byPeriod //原樣帶入(有效性由dataConvert檢核)
                    }
                    continue
                }
                need = true
            }
        }
        if (need) {
            log(`計算nm: ${def.id}`)
            let methodsProbe = map(methodsRaw, (m) => ({
                method: m.method,
                optMethod: withIv(m.optMethod),
                deals: map(m.deals, (d) => ({ keyIn: d.keyIn, keyOut: d.keyOut, nm: { avg: 0, std: 1 } })),
            }))
            let rs = await wdt.dataConvert(cfg.name, cfg.symbol, cfg.interval, def.cointpe, def.keyKind, methodsProbe, arrNm, kp, fdParam, {
                funAddKeysIn: def.funAddKeysIn,
                keysIn: def.keysIn,
                denyOutput: true,
            })
            for (let r of rs) {
                let nk = nmKeyOf(def, r.method, r.keyIn, r.keyOut)
                if (haskey(meta.nm, nk)) {
                    continue
                }
                if (nmUseNew && !Array.isArray(r.periods)) {
                    throw new Error(`nmScope 'period' and nmStat 'medianIqr' need a newer w-data-trade whose dataConvert returns periods, median, q25 and q75`)
                }
                //鍵層(各期別攤平): nmStat為'medianIqr'取中位數與四分位距, 不可用則退回平均與標準差, 再不行為{avg:0,std:1}(現行之取法)
                let nmMi = nmStat === 'medianIqr' ? nmMedianIqrOf(r) : null
                let avg = 0
                let std = 1
                if (nmMi) {
                    avg = nmMi.avg
                    std = nmMi.std
                }
                else {
                    if (nmStat === 'medianIqr') {
                        log(`  警告: ${nk} 之中位數與四分位距不可用, nm改採平均與標準差`)
                    }
                    avg = isnum(r.avg) && Number.isFinite(r.avg) ? r.avg : 0
                    std = isnum(r.std) && Number.isFinite(r.std) && r.std > 0 ? r.std : 1
                    if (!(isnum(r.std) && Number.isFinite(r.std) && r.std > 0)) {
                        log(`  警告: ${nk} 之std非有限正數(${r.std}), nm改採{avg:${avg},std:${std}}`) //訊息之avg與std為實際採用者(僅log文字, 不影響寫出之nm; avg有限時保留之, 不改為0)
                    }
                }
                meta.nm[nk] = { avg, std }
                if (nmScope === 'period') {
                    meta.nm[nk].byPeriod = nmByPeriodOf(r.periods, nmStat) //頂層為鍵層之值(無對應期別時之後備), byPeriod只收n至少NM_MIN_N且尺度大於0之期別
                }
                nmComputed.push({ key: nk, avg, std, min: r.min, max: r.max })
            }
        }

        //methods, 掛上凍結之nm
        let methods = map(methodsRaw, (m) => ({
            method: m.method,
            optMethod: withIv(m.optMethod),
            deals: map(m.deals, (d) => {
                let nk = nmKeyOf(def, m.method, d.keyIn, d.keyOut)
                let nm = get(meta.nm, nk, null)
                if (!iseobj(nm)) {
                    throw new Error(`nm缺失[${nk}](計算結果無此組合, 請確認定義之keyIn/keyOut與計算函數輸出一致)`)
                }
                return { keyIn: d.keyIn, keyOut: d.keyOut, nm }
            }),
        }))

        //write
        log(`寫出序列: ${def.id}`)
        await wdt.dataConvert(cfg.name, cfg.symbol, cfg.interval, def.cointpe, def.keyKind, methods, arrSrc, kp, fdParam, {
            funAddKeysIn: def.funAddKeysIn,
            keysIn: def.keysIn,
        })
    }

    //series, 外部序列直接寫入
    let series = get(opt, 'series', null)
    let seriesKeys = []
    if (iseobj(series)) {
        for (let key of Object.keys(series)) {
            if (!isestr(key) || /[\\/]/.test(key)) {
                throw new Error(`opt.series之key[${key}]無效`)
            }
            let arr = series[key]
            if (isfun(arr)) {
                arr = await arr(cfg, key)
            }
            if (!isearr(arr)) {
                throw new Error(`opt.series.${key}須為非空陣列[{time,param}]`)
            }
            for (let i = 0; i < arr.length; i++) {
                if (!istime(get(arr[i], 'time', '')) || !haskey(arr[i], 'param')) {
                    throw new Error(`opt.series.${key}[${i}]須為{time:'YYYY-MM-DDTHH:mm:ss',param}`)
                }
            }
            writeJson(path.resolve(fdParam, `${key}.json`), map(arr, (v) => ({ time: v.time, param: v.param })), { format: false })
            seriesKeys.push(key)
        }
    }

    //meta
    meta.nmPeriod = nmPeriod
    if (nmUseNew) {
        meta.nmScope = nmScope //兩者皆為預設時不寫入(使既有工作根之meta/params.json逐位元組不變)
        meta.nmStat = nmStat
    }
    meta.name = cfg.name
    meta.interval = cfg.interval
    meta.useIndex = cfg.useIndex
    meta.defs = map(defs, 'id')
    meta.dropped = dropped
    meta.seriesKeys = seriesKeys
    meta.at = nowIso()
    writeJson(fpMeta, meta)

    let keys = listJsonKeys(fdParam)
    return {
        nKeys: keys.length,
        keys,
        defs: map(defs, 'id'),
        dropped,
        calibrated,
        nmComputed: nmComputed.length,
        meta,
        nmScope,
        nmStat,
    }
}
